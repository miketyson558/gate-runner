];

function artSvg(t){

 n=+$('difficulty').value;pieces=[];selected=-1;moves=0;seconds=0;started=true;finished=false;showPreview=false;clearInterval(timer);

 // Avoid a solved board by swapping if random order accidentally aligns.
 if(order.every((v,i)=>v===i))[order[0],order[1]]=[order[1],order[0]];

  ctx.save();ctx.beginPath();ctx.rect(x+2,y+2,s-4,s-4);ctx.clip();

  ctx.restore();

 });
}
function slotFromEvent(e){const r=canvas.getBoundingClientRect();let x=(e.clientX-r.left)*560/r.width,y=(e.clientY-r.top)*560/r.height;return Math.max(0,Math.min(n-1,Math.floor(y/(560/n))))*n+Math.max(0,Math.min(n-1,Math.floor(x/(560/n))))}
